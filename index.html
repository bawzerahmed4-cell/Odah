function switchTab(tabId) {

document.querySelectorAll('.tab-btn').forEach(el => { el.classList.remove('bg-blue-600', 'text-white'); el.classList.add('text-slate-300', 'hover:bg-slate-800'); });

}

function formatPrice(priceInSAR) {
const info = exchangeRates[currentCurrency];
return `${(priceInSAR * info.rate).toLocaleString('en-US', {maximumFractionDigits: 0})} ${info.symbol}`;
}

function renderProducts() {
const grid = document.getElementById('productsGrid');
const search = document.getElementById('marketSearch') ? document.getElementById('marketSearch').value.toLowerCase() : '';
grid.innerHTML = '';

const filtered = products.filter(p => p.name.toLowerCase().includes(search) || p.category.toLowerCase().includes(search) || p.storeName.toLowerCase().includes(search));

if (filtered.length === 0) {
grid.innerHTML = '<div class="col-span-full text-center py-10 text-slate-400">لا توجد قطع غيار معروضة في السوق حالياً.</div>';
return;
}

filtered.forEach(p => {
grid.innerHTML += `
<div class="bg-cardBg p-5 rounded-2xl border border-slate-800 flex flex-col justify-between space-y-4 shadow-lg">
<div class="flex justify-between items-start">
<div>
<span class="text-xs bg-blue-600/20 text-blue-400 px-2.5 py-1 rounded-md font-semibold">${p.category}</span>
<h3 class="font-bold text-lg mt-2">${p.name}</h3>
</div>
<img src="${p.storeLogo || merchantProfile.logo}" class="w-10 h-10 rounded-full object-cover border border-amber-500/30" title="${p.storeName}">
</div>
<p class="text-xs text-slate-400">المتجر: <span class="text-slate-200 font-semibold">${p.storeName}</span> | المتوفر: <span class="text-slate-200 font-semibold">${p.stock} قطعة</span></p>
<div class="flex items-center justify-between pt-2 border-t border-slate-800">
<span class="font-bold text-lg text-amber-400">${formatPrice(p.priceSAR)}</span>
<button onclick="addToCart('${p.name}', ${p.priceSAR}, '${p.storeName}')" class="bg-blue-600 hover:bg-blue-500 text-white text-xs px-3 py-2 rounded-xl">➕ إضافة للسلة</button>
</div>
</div>
`;
});
}

function addToCart(name, priceSAR, storeName) {
cart.push({ name, priceSAR, storeName });
document.getElementById('cartBadge').innerText = cart.length;
alert(`تم إضافة "${name}" إلى السلة`);
}

function renderCart() {
const list = document.getElementById('cartItemsList');
list.innerHTML = cart.length === 0 ? '<p class="text-slate-400 text-sm text-center py-6">السلة فارغة حالياً.</p>' : '';
let totalSAR = 0;
cart.forEach((item, idx) => {
totalSAR += item.priceSAR;
list.innerHTML += `<div class="bg-slate-900 p-3 rounded-xl border border-slate-700 flex justify-between items-center text-sm"><span>${item.name} - ${item.storeName} (${formatPrice(item.priceSAR)})</span><button onclick="cart.splice(${idx},1);renderCart()" class="text-red-400 text-xs">حذف</button></div>`;
});
document.getElementById('cartTotalPrice').innerText = formatPrice(totalSAR);
}

function checkoutCartWhatsApp() {
if (cart.length === 0) return;
let totalSAR = cart.reduce((sum, item) => sum + item.priceSAR, 0);
let text = cart.map((item, idx) => `${idx+1}. ${item.name} (${item.storeName}) - ${formatPrice(item.priceSAR)}`).join('\n');

cart.forEach(item => sales.push({ productName: item.name, priceSAR: item.priceSAR, date: new Date().toISOString().split('T')[0] }));
localStorage.setItem('odda_sales_clean', JSON.stringify(sales));

window.open(`https://wa.me/?text=${encodeURIComponent(`طلب من منصة عُدّة:\n\n${text}\n\nالإجمالي: ${formatPrice(totalSAR)}`)}`, '_blank');
cart = []; document.getElementById('cartBadge').innerText = 0; renderCart();
}

function saveNewProduct() {
const name = document.getElementById('newProdName').value;
const category = document.getElementById('newProdCategory').value;
const price = parseFloat(document.getElementById('newProdPrice').value);
const stock = parseInt(document.getElementById('newProdStock').value) || 1;
if(!name || !price) { alert('أدخل اسم القطعة والسعر'); return; }

products.push({
id: Date.now().toString(),
name,
category: category || 'عام',
priceSAR: price,
stock,
storeName: merchantProfile.name,
storeLogo: merchantProfile.logo
});

localStorage.setItem('odda_products_clean', JSON.stringify(products));
renderProducts();
renderMerchantProducts();
closeModal('productModal');
}

function renderMerchantProducts() {
const list = document.getElementById('merchantProductsList');
list.innerHTML = '';
if(products.length === 0) { list.innerHTML = '<p class="text-slate-400 text-sm">لم تقم بإضافة أي قطعة غيار بعد.</p>'; return; }

products.forEach(p => {
list.innerHTML += `<div class="bg-slate-900 p-4 rounded-xl border border-slate-700 flex justify-between items-center text-sm">
<div><h4 class="font-bold">${p.name}</h4><p class="text-xs text-slate-400">المخزون: ${p.stock} | السعر: ${formatPrice(p.priceSAR)}</p></div>
<button onclick="deleteProduct('${p.id}')" class="bg-red-600/20 text-red-400 px-3 py-1.5 rounded-lg text-xs">حذف</button>
</div>`;
});
document.getElementById('statMyProductsCount').innerText = products.length;
}

function deleteProduct(id) {
products = products.filter(p => p.id !== id);
localStorage.setItem('odda_products_clean', JSON.stringify(products));
renderProducts();
renderMerchantProducts();
renderSuperAdminView();
}

function renderSuperAdminView() {

<span><strong>${p.name}</strong> (${p.storeName}) - ${p.priceSAR} SAR</span>
<button onclick="deleteProduct('${p.id}')" class="bg-red-600 text-white px-2 py-1 rounded">حذف المالك (حظر)</button>
</div>`;
});
}
